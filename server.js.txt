const express = require('express');
const cors = require('cors');
const fs = require('fs');
const path = require('path');

const app = express();
const PORT = process.env.PORT || 80;

app.use(express.json());
app.use(cors());

const KEYS_FILE = path.join(__dirname, 'licenses.json');

function getLicenses() {
    if (!fs.existsSync(KEYS_FILE)) {
        fs.writeFileSync(KEYS_FILE, JSON.stringify([]));
    }
    try {
        const data = fs.readFileSync(KEYS_FILE, 'utf8');
        return JSON.parse(data);
    } catch {
        return [];
    }
}

function saveLicenses(licenses) {
    fs.writeFileSync(KEYS_FILE, JSON.stringify(licenses, null, 2));
}

// استضافة مجلد التحديثات
app.use('/launcher', express.static(path.join(__dirname, 'public')));

// 1. API التحقق من المفتاح (للانشر)
app.post('/api/verify-key', (req, res) => {
    const { license_key } = req.body;
    let licenses = getLicenses();
    const found = licenses.find(k => k.key === license_key && k.active === true);
    
    if (found) {
        return res.json({ success: true, message: "License is active" });
    } else {
        return res.json({ success: false, message: "Invalid or expired key" });
    }
});

// 2. API جلب المفاتيح (للداشبورد)
app.get('/api/keys', (req, res) => {
    res.json(getLicenses());
});

// 3. API إضافة مفتاح
app.post('/api/keys/add', (req, res) => {
    const { key, note } = req.body;
    if (!key) return res.status(400).json({ error: 'Key is required' });
    
    let licenses = getLicenses();
    if (!licenses.some(k => k.key === key.trim())) {
        licenses.push({ key: key.trim(), active: true, note: note ? note.trim() : '' });
        saveLicenses(licenses);
        return res.json({ success: true });
    } else {
        return res.status(400).json({ error: 'Key already exists' });
    }
});

// 4. API تعديل أو حذف مفتاح
app.post('/api/keys/action', (req, res) => {
    const { key, action } = req.body;
    let licenses = getLicenses();
    
    if (action === 'delete') {
        licenses = licenses.filter(k => k.key !== key);
    } else if (action === 'toggle') {
        let item = licenses.find(k => k.key === key);
        if (item) item.active = !item.active;
    }
    
    saveLicenses(licenses);
    return res.json({ success: true });
});

app.listen(PORT, () => {
    console.log(`Server is running on port ${PORT}`);
});
