            <button class="btn-add" onclick="addKey()">إضافة المفتاح</button>
        </div>

        <h3>المفاتيح المسجلة حالياً</h3>
        <table>
            <thead>
                <tr>
                    <th>مفتاح الترخيص</th>
                    <th>الحالة</th>

                </tr>
            </thead>
            <tbody id="keysTable">
                <!-- البيانات تظهر هنا -->
            </tbody>

    <script>
        let keys = JSON.parse(localStorage.getItem('cf_keys')) || [
            { key: "CF-DEMO-2026", active: true, note: "مفتاح تجريبي" }
        ];

        function renderTable() {
            let tbody = document.getElementById('keysTable');
            if (keys.length === 0) {
                tbody.innerHTML = '<tr><td colspan="4">لا توجد مفاتيح مسجلة حتى الآن</td></tr>';
                return;
            }

            tbody.innerHTML = keys.map((k, index) => `
                <tr>
                    <td><b>${k.key}</b></td>
                    <td><span class="${k.active ? 'status-active' : 'status-disabled'}">${k.active ? 'مفعل' : 'معطل'}</span></td>
                    <td>${k.note || '-'}</td>
                    <td>
                        <button class="btn-toggle" onclick="toggleKey(${index})">تبديل</button>
                        <button class="btn-delete" onclick="deleteKey(${index})">حذف</button>
                    </td>
                </tr>
            `).join('');
        }

        function addKey() {
            let keyInput = document.getElementById('newKey');
            let noteInput = document.getElementById('newNote');
            let key = keyInput.value.trim();
            let note = noteInput.value.trim();

            if (!key) return alert('الرجاء كتابة مفتاح الترخيص أولاً!');
            if (keys.some(k => k.key === key)) return alert('هذا المفتاح موجود مسبقاً!');

            keys.push({ key, active: true, note });
            localStorage.setItem('cf_keys', JSON.stringify(keys));
            
            keyInput.value = '';
            noteInput.value = '';
            renderTable();
        }

        function toggleKey(index) {
            keys[index].active = !keys[index].active;
            localStorage.setItem('cf_keys', JSON.stringify(keys));
            renderTable();
        }

        function deleteKey(index) {
            keys.splice(index, 1);
            localStorage.setItem('cf_keys', JSON.stringify(keys));
            renderTable();
        }

        renderTable();
    </script>
